{{- /* Overrides Blowfish's schema.html to: - Add a standalone Person node on the homepage with sameAs social links - Emit BlogPosting (not generic Article) for the posts/portfolio sections - Include an image property on articles when a featured image exists */ -}} {{- $iso8601 := "2006-01-02T15:04:05-07:00" -}} {{- if .IsHome -}} {{- $sameAs := slice -}} {{- with .Site.Params.Author.links -}} {{- range . -}} {{- range $k, $v := . -}}{{- $sameAs = $sameAs | append $v -}}{{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $schemaType := "Article" -}} {{- if in (slice "posts" "post" "portfolio") .Section -}} {{- $schemaType = "BlogPosting" -}} {{- end -}} {{- $image := "" -}} {{- with .Resources.GetMatch "featured*" -}}{{- $image = .Permalink -}}{{- end -}} {{- if not $image -}} {{- with .Params.images -}}{{- with index . 0 -}}{{- $image = . | absURL -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}}
